⚙ Google Apps Script Setup

Set this up once for live prices on desktop & mobile.
1Go to script.google.com → New Project
2Paste this code:
function doGet(e){ var html=UrlFetchApp.fetch('https://www.merolagani.com/LatestMarket.aspx', {muteHttpExceptions:true,headers:{'User-Agent':'Mozilla/5.0'}}).getContentText(); var prices={}; var rows=html.match(/<tr[\s\S]*?<\/tr>/gi)||[]; rows.forEach(function(row){ var cells=row.match(/<td[^>]*>([\s\S]*?)<\/td>/gi)||[]; if(cells.length<4)return; var g=function(c){return c.replace(/<[^>]+>/g,'').trim();}; var sym=g(cells[0]).toUpperCase().replace(/\s/g,''); var ltp=parseFloat(g(cells[1]).replace(/,/g,'')); if(sym&&ltp>0&&sym.length>=2&&sym.length<=12) prices[sym]={ltp:ltp,chg:g(cells[2])}; }); return ContentService.createTextOutput(JSON.stringify(prices)) .setMimeType(ContentService.MimeType.JSON); }

    📋 JSON Format Guide & Examples ▾
    

      
Format 1 — Simple holdings array:

      
[
  {"sym": "NABIL", "name": "Nabil Bank", "qty": 455, "wacc": 433.06, "sector": "Commercial Banks"},
  {"sym": "NICA",  "name": "NIC Asia",   "qty": 739, "wacc": 436.00, "sector": "Commercial Banks"}
]

      
Format 2 — Full portfolio with transactions:

      
{
  "portfolio": "My Portfolio",
  "description": "Long-term",
  "holdings": [
    {"sym":"NABIL","qty":455,"wacc":433.06,"sector":"Commercial Banks"}
  ],
  "transactions": [
    {"type":"BUY","sym":"NABIL","qty":455,"price":433.06,"date":"2024-01-15","brokerage":120}
  ]
}

      
Format 3 — NEPSE broker / MeroShare export style:

      
[
  {"Script": "NABIL", "Quantity": 455, "Rate": 433.06},
  {"Script": "NICA",  "Quantity": 739, "Rate": 436.00}
]

      
Accepted field aliases: sym/script/symbol/scrip · qty/quantity/shares · wacc/rate/price/cost · name/company · sector/type · date/transactionDate
